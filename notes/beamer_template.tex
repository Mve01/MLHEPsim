\section{Introduction}

\begin{frame}{Motivation}
  \begin{itemize}
    \item Current MC simulation slow
    \item Need for fast generation of events
    \item Potential of normalizing flows
    \item Method widely applicable to other processes and parameters
  \end{itemize}
\end{frame}

\begin{frame}{Motivation: MC simulation slow}
    \begin{itemize}
    \item In 2017 20\% of the ATLAS computing resources were used for MC simulation \cite{hsf2021mcgen}
    \item Current MC generation can take days or weeks for complex processes
    \item With the HL-LHC, more complex events and more data will require even more MC simulation
    \item We need faster and more efficient MC generation
    \end{itemize}
    
\end{frame}

\section{Idea}
\begin{frame}{Idea}
  \begin{itemize}
    \item Take already existing MC data
    \item Train a conditional normalizing flow to learn the distribution of events
    \item Condition the flow on a Wilson coefficient 
    \item Use the trained flow to generate new events for any value of the Wilson coefficient, much faster than traditional MC simulation
  \end{itemize}
\end{frame}

\section{Setup}
\begin{frame}{Normalizing flow: MAFMADEMOG}
  \begin{itemize}
    \item Masked Autoregressive Flow (MAF) 
    \begin{itemize}
      \item A type of normalizing flow that models complex distributions by transforming a simple base distribution through a series of invertible transformations
    \end{itemize}
    \item MADE (Masked Autoencoder for Distribution Estimation) 
    \begin{itemize}
      \item A type of autoregressive model that allows for efficient sampling and density estimation
    \end{itemize}
    \item MOG (Mixture of Gaussians) 
    \begin{itemize}
      \item Used as the base distribution for the flow, allowing it to model complex distributions
    \end{itemize}
  \end{itemize}

\end{frame}
